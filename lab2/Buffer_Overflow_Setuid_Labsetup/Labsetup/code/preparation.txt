# Also compute return-slot and exact distance to return address.
mkdir -p /home/seed/Documents/Laboratoire/lab1_v2/lab2/traces
gdb -q -batch \
	-ex 'set pagination off' \
	-ex 'b bof' \
	-ex 'run' \
	-ex 'next' \
	-ex 'printf "E1_EBP=0x%x\n", $ebp' \
	-ex 'printf "E1_BUFFER=0x%x\n", &buffer' \
	-ex 'printf "E1_RET_SLOT=0x%x\n", $ebp+4' \
	-ex 'printf "E1_DISTANCE=%d\n", (int)((char*)($ebp+4) - (char*)&buffer)' \
	-ex 'p/x $ebp' \
	-ex 'p/x &buffer' \
	-ex 'p/x $ebp+4' \
	-ex 'p/d (int)((char*)($ebp+4) - (char*)&buffer)' \
	-ex 'x/2wx $ebp' \
	./stack-L1-dbg | tee /home/seed/Documents/Laboratoire/lab1_v2/lab2/traces/T1-stack-layout.log

# Print readable mapping in terminal so values are directly interpretable.
E1_EBP=$(grep -m1 '^E1_EBP=' /home/seed/Documents/Laboratoire/lab1_v2/lab2/traces/T1-stack-layout.log | cut -d= -f2)
E1_BUFFER=$(grep -m1 '^E1_BUFFER=' /home/seed/Documents/Laboratoire/lab1_v2/lab2/traces/T1-stack-layout.log | cut -d= -f2)
E1_RET_SLOT=$(grep -m1 '^E1_RET_SLOT=' /home/seed/Documents/Laboratoire/lab1_v2/lab2/traces/T1-stack-layout.log | cut -d= -f2)
E1_DISTANCE=$(grep -m1 '^E1_DISTANCE=' /home/seed/Documents/Laboratoire/lab1_v2/lab2/traces/T1-stack-layout.log | cut -d= -f2)

echo
echo '========== E1 INTERPRETATION =========='
echo "buffer address      (&buffer)   = ${E1_BUFFER}"
echo "frame pointer       (ebp)       = ${E1_EBP}"
echo "return-address slot (ebp + 4)   = ${E1_RET_SLOT}"
echo "offset to overwrite RET         = ${E1_DISTANCE} bytes"
echo
cat <<'EOF'
Terminology and why it matters for exploitation:
- buffer: local array in bof(); overflow starts here.
- ebp (frame pointer): anchor for current stack frame.
  In 32-bit calling convention:
  * [ebp]     holds saved caller EBP
  * [ebp + 4] holds saved return address (RET target)
- return-address slot: value CPU jumps to when bof() executes ret.

How we use this for the attack:
1) Fill payload with NOP/shellcode and padding.
2) At exact offset "distance" (here E1_DISTANCE), overwrite saved RET.
3) Set overwritten RET to point into our payload (NOP sled/shellcode).
4) When bof() returns, control flow is redirected to attacker-controlled code.
EOF
